    <script>
        // ROLAGEM DO MENU SUPERIOR
        function rolarMenu(distancia) {
            const menu = document.getElementById('menu-scroll');
            if (menu) {
                menu.scrollBy({ left: distancia, behavior: 'smooth' });
            }
        }

        // LÓGICA DO CARROSSEL / BANNER HERO
        const slides = [
            {
                subtitulo: "Ponto central de informações",
                titulo: "Programa Reforma Tributária",
                descricao: "Acompanhe marcos, entregas e treinamentos, além de acessar materiais exclusivos e indicadores de evolução.",
                imagem: "https://images.unsplash.com/photo-1573496359142-b8d87734a5a2?q=80&w=1600&auto=format&fit=crop"
            },
            {
                subtitulo: "Acompanhamento em Tempo Real",
                titulo: "Relatórios e Análises Estratégicas",
                descricao: "Consulte painéis interativos e faça o upload de planilhas para visualizar os impactos tributários diretamente no gráfico.",
                imagem: "https://images.unsplash.com/photo-1460925895917-afdab827c52f?q=80&w=1600&auto=format&fit=crop"
            }
        ];

        let slideAtual = 0;

        function atualizarBanner() {
            const slide = slides[slideAtual];
            document.getElementById('banner-img').src = slide.imagem;
            document.getElementById('banner-subtitulo').innerText = slide.subtitulo;
            document.getElementById('banner-titulo').innerText = slide.titulo;
            document.getElementById('banner-descricao').innerText = slide.descricao;

            // Atualiza os pontos (dots) de indicação
            slides.forEach((_, idx) => {
                const dot = document.getElementById(`dot-${idx}`);
                if (dot) {
                    if (idx === slideAtual) {
                        dot.className = "h-2.5 w-2.5 rounded-full bg-sky-500 cursor-pointer transition-all";
                    } else {
                        dot.className = "h-2.5 w-2.5 rounded-full bg-slate-300 hover:bg-slate-400 cursor-pointer transition-all";
                    }
                }
            });
        }

        function mudarSlide(direcao) {
            slideAtual += direcao;
            if (slideAtual < 0) {
                slideAtual = slides.length - 1;
            } else if (slideAtual >= slides.length) {
                slideAtual = 0;
            }
            atualizarBanner();
        }

        function irParaSlide(index) {
            slideAtual = index;
            atualizarBanner();
        }

        // GRÁFICO CHART.JS
        const ctx = document.getElementById('meuGraficoLinhas').getContext('2d');
        const meuGrafico = new Chart(ctx, {
            type: 'line',
            data: {
                labels: ['Item 1', 'Item 2', 'Item 3', 'Item 4', 'Item 5'],
                datasets: [{
                    label: 'Valores Processados',
                    data: [0, 0, 0, 0, 0],
                    borderColor: '#0284c7',
                    backgroundColor: 'rgba(2, 132, 199, 0.1)',
                    borderWidth: 3,
                    fill: true,
                    tension: 0.4,
                    pointBackgroundColor: '#0284c7',
                    pointRadius: 4
                }]
            },
            options: {
                responsive: true,
                maintainAspectRatio: false,
                plugins: { legend: { display: false } },
                scales: {
                    x: { grid: { display: false } },
                    y: { border: { dash: [5, 5] }, grid: { color: '#f1f5f9' } }
                }
            }
        });

        function atualizarNomeFicheiro(event) {
            const file = event.target.files[0];
            if (file) {
                document.getElementById('nome-ficheiro').innerText = file.name;
            }
        }

        function processarFicheiroExcel() {
            const input = document.getElementById('input-excel');
            const file = input.files[0];

            if (!file) {
                alert('Por favor, selecione primeiro um ficheiro Excel (.xlsx ou .csv)!');
                return;
            }

            const btn = document.getElementById('btn-upload');
            
            btn.disabled = true;
            btn.className = "bg-sky-500 text-white text-xs font-semibold px-4 py-2 rounded-lg shadow-sm flex items-center justify-center gap-2 opacity-80 cursor-not-allowed shrink-0";
            btn.innerHTML = `
                <svg class="animate-spin h-4 w-4 text-white shrink-0" xmlns="http://www.w3.org/2000/svg" fill="none" viewBox="0 0 24 24">
                    <circle class="opacity-25" cx="12" cy="12" r="10" stroke="currentColor" stroke-width="4"></circle>
                    <path class="opacity-75" fill="currentColor" d="M4 12a8 8 0 018-8V0C5.373 0 0 5.373 0 12h4zm2 5.291A7.962 7.962 0 014 12H0c0 3.042 1.135 5.824 3 7.938l3-2.647z"></path>
                </svg>
                <span>A processar...</span>
            `;

            const reader = new FileReader();

            reader.onload = function(e) {
                const data = new Uint8Array(e.target.result);
                const workbook = XLSX.read(data, { type: 'array' });
                const primeirAba = workbook.Sheets[workbook.SheetNames[0]];
                const dadosJson = XLSX.utils.sheet_to_json(primeirAba, { header: 1 });

                const novosRotulos = [];
                const novosValores = [];
                let somaTotal = 0;

                for (let i = 1; i < dadosJson.length; i++) {
                    const linha = dadosJson[i];
                    if (linha.length >= 2) {
                        novosRotulos.push(String(linha[0]));
                        const valor = Number(linha[1]) || 0;
                        novosValores.push(valor);
                        somaTotal += valor;
                    }
                }

                setTimeout(() => {
                    meuGrafico.data.labels = novosRotulos;
                    meuGrafico.data.datasets[0].data = novosValores;
                    meuGrafico.update();

                    document.getElementById('kpi-total').innerText = 'R$ ' + somaTotal.toLocaleString('pt-BR');
                    document.getElementById('kpi-linhas').innerText = novosValores.length;
                    document.getElementById('badge-status').innerText = 'Atualizado via Excel';

                    btn.disabled = false;
                    btn.className = "bg-sky-500 hover:bg-sky-600 text-white text-xs font-semibold px-4 py-2 rounded-lg transition-all shadow-sm flex items-center justify-center shrink-0";
                    btn.innerHTML = `<span>Processar</span>`;
                }, 1000);
            };

            reader.readAsArrayBuffer(file);
        }
    </script>
